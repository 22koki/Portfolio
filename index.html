# Spatial workflow concept
import geopandas as gpd
from sklearn.ensemble import RandomForestClassifier

features = gpd.read_file("spatial_features.geojson")
features = features.to_crs("EPSG:32737")

X = features[["access", "density", "elevation"]]
model = RandomForestClassifier(n_estimators=100)
model.fit(X, features["class"])

features["prediction"] = model.predict(X)
